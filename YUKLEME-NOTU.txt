import json
import os
from datetime import datetime, timezone
from pathlib import Path
from urllib.parse import urlencode
from urllib.request import urlopen

TOKEN = os.environ.get("INSTAGRAM_ACCESS_TOKEN", "").strip()
if not TOKEN:
    raise SystemExit("INSTAGRAM_ACCESS_TOKEN secret is missing")

fields = "id,caption,media_type,media_url,thumbnail_url,permalink,timestamp"
query = urlencode({"fields": fields, "limit": 12, "access_token": TOKEN})
url = f"https://graph.instagram.com/me/media?{query}"

with urlopen(url, timeout=30) as response:
    payload = json.load(response)

posts = []
for item in payload.get("data", []):
    if not item.get("permalink"):
        continue
    posts.append({key: item.get(key) for key in (
        "id", "caption", "media_type", "media_url", "thumbnail_url", "permalink", "timestamp"
    )})

output = {
    "updated_at": datetime.now(timezone.utc).isoformat(),
    "posts": posts[:6],
}
Path("assets/instagram-feed.json").write_text(
    json.dumps(output, ensure_ascii=False, indent=2), encoding="utf-8"
)
print(f"Updated Instagram feed with {len(output['posts'])} posts")
